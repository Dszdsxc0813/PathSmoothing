# path_corrector.py
import math
import numpy as np
from shapely.geometry import Point
from collections import namedtuple


class PathCorrector:
    def __init__(self, turn_radius):
        self.turn_radius = turn_radius  # R

    def _apply_correction(self, robot, segment_start, segment_end, D_ac_expected, D_ac_actual, segment_next_end):

        """
            完整路径矫正流程：
            1) 圆O2'弧 Z->H
            2) 圆O3  弧 H->X
            3) 直线   X->N
            4) 圆O4  弧 N->Z2
            返回整个矫正路径点列表
        """

        """论文公式6-9的实现：计算矫正路径的圆心和交点"""
        # 提取当前机器人状态
        current_pos = robot.current_pos
        alpha = robot.heading_angle  # 当前朝向角
        R = self.turn_radius

        # --- 1) 计算 O2_center、H ---
        # 公式6-7：计算圆心 O2 坐标
        x_O2 = current_pos.x + R * math.cos(alpha)
        y_O2 = current_pos.y + R * math.sin(alpha)
        O2_center = Point(x_O2, y_O2)

        # 公式8：直线 L 的计算
        x1, y1 = segment_start.x, segment_start.y
        x2, y2 = segment_end.x, segment_end.y
        dx = x2 - x1
        dy = y2 - y1
        # 避免除以 0
        if dx == 0:
            k = float('inf')  # 垂直线
        else:
            k = dy / dx

        if k != float('inf'):
            offset = R / math.sqrt(1 + k ** 2) * (k + 1)

            # y = k(x - x1) + y1 - offset
            def line_L(x):
                return k * (x - x1) + y1 - offset
        else:
            # 如果是垂直线，x = x1，y 平移 offset
            def line_L(x):
                return y1 - R * (1 + 1)  # k + 1 -> 1 + 1，k -> inf 用 1 近似

        # 公式9：以O2为中心，2R为半径的圆方程
        def circle_O2(x, y):
            return (x - O2_center.x) ** 2 + (y - O2_center.y) ** 2 - (2 * R) ** 2

        # 公式10-14：计算直线L与圆O2的交点
        A_quad = 1 + k ** 2  # A = 1 + k^2

        # 预先准备参数
        x1, y1 = segment_start.x, segment_start.y
        x_O2, y_O2 = O2_center.x, O2_center.y

        # 公式13：计算 u（对应你代码中的 B）
        u = (-2 * x_O2
             - 2 * k ** 2 * x1
             - 2 * k ** 2 * R
             + 2 * k * y1
             - 2 * k * R
             - 2 * k * y_O2)

        # 公式14：计算 v（对应你代码中的 C）
        v = (k ** 2 * x1 ** 2
             + 2 * k ** 2 * R * x1
             - 2 * k * y1 * x1
             + 2 * k * R * x1
             + 2 * k * y_O2 * x1
             + k ** 2 * R ** 2
             - 2 * k * R * y1
             + 2 * k * R * y_O2
             + y1 ** 2
             - 2 * R * y1
             - 2 * y_O2 * y1
             + 2 * k * R ** 2
             + R ** 2
             + 2 * y_O2 * R
             + y_O2 ** 2
             + x_O2 ** 2
             - 4 * R ** 2)

        # 判别式
        discriminant = u ** 2 - 4 * A_quad * v
        if discriminant < 0:
            return []  # 无解时返回空路径

        # 公式10-11：求两个交点的横坐标
        sqrt_dis = math.sqrt(discriminant)
        x_a = (-u + sqrt_dis) / (2 * A_quad)
        x_b = (-u - sqrt_dis) / (2 * A_quad)

        # --- 3) 公式15-16：选出圆心 O3 ---
        if (x_a - x_b) / (x2 - x1) > 0:
            x_O3 = x_a
        else:
            x_O3 = x_b

        y_O3 = k * (x_O3 - x1) + y1 - offset  # 公式16
        O3_center = Point(x_O3, y_O3)

        # --- 4) 终点 X：联立圆 O3 和 AB 直线 ---
        # 圆 O3: (x - x_O3)^2 + (y - y_O3)^2 = R^2
        # 直线 AB: y = k*(x - x1) + y1
        A_x = 1 + (0 if k == float('inf') else k ** 2)
        B_x = -2 * x_O3 + 2 * k * (y1 - y_O3 - k * x1)
        C_x = (
                x_O3 ** 2
                + k ** 2 * x1 **2
                - 2 * k * x1 * y1
                + 2 * k * y_O3 * x1
                + y1 ** 2
                - 2 * y_O3 * y1
                +y_O3 ** 2
                - R ** 2
        )

        disc2 = B_x ** 2 - 4 * A_x * C_x
        if disc2 < 0:
            return O2_center, O3_center, []
        sqrt2 = math.sqrt(disc2)
        x_X1 = (-B_x + sqrt2) / (2 * A_x)
        x_X2 = (-B_x - sqrt2) / (2 * A_x)
        y_X1 = k * (x_X1 - x1) + y1
        y_X2 = k * (x_X2 - x1) + y1

        if x_X1 == x_X2 and y_X1 == y_X2:
            Point(x_X1, y_X1)
        else:
            print("矫正路径终点X存在多个值，请修改代码逻辑！")

        # ——— 5) 计算 O2' 与 O3 的交点 H ———
        # O2'是以O2为圆心，以R为半径的小圆
        H = self.compute_circle_tangent_point(O2_center, O3_center, self.R)
        print(f"圆弧过渡切点 H 坐标: ({H.x:.2f}, {H.y:.2f})")

        # ———— 6) 计算下一段路径的预期转向点N
        N = self.compute_blend_start(segment_start, segment_end, segment_next_end, self.turn_radius)

    """
        计算两个圆（半径相同 R，圆心分别为 O2 和 O3）唯一的外切点 H
    """
    def compute_circle_tangent_point(self, O2_center, O3_center, R):

        x1, y1 = O2_center.x, O2_center.y
        x2, y2 = O3_center.x, O3_center.y

        # 圆心连线方向向量
        dx = x2 - x1
        dy = y2 - y1
        dist = math.hypot(dx, dy)

        if dist == 0:
            raise ValueError("O2 and O3 centers are identical — no unique tangent point.")

        # 单位方向向量
        ux = dx / dist
        uy = dy / dist

        # 从 O2 沿着方向向量移动 R 的距离
        x_H = x1 + R * ux
        y_H = y1 + R * uy

        return Point(x_H, y_H)

    """
    计算两条直线 AB、BC 的内侧平移线（偏移距离 = turn_radius），
    求它们的交点 —— 也就是平滑转弯圆的圆心 O_blend，
    再在原始直线 AB 上，找出与 O_blend 距离恰好为 turn_radius 的点 —— 这就是预期转向起点 N。
    """
    def compute_blend_start(self, A: Point, B: Point, C: Point, R: float) -> Point:
        """
        给定三点 A→B→C 以及转弯半径 R，
        返回位于 AB 上、到两直线 AB/BC 垂线距离均为 R 的点 N。
        """

        def to_vec(P, Q):
            return (Q.x - P.x, Q.y - P.y)

        def norm(v):
            return math.hypot(v[0], v[1])

        def scale(v, s):
            return (v[0] * s, v[1] * s)

        def add(u, v):
            return (u[0] + v[0], u[1] + v[1])

        def dot(u, v):
            return u[0] * v[0] + u[1] * v[1]

        def cross(u, v):
            return u[0] * v[1] - u[1] * v[0]

        # 1) 计算 AB 和 BC 的单位方向向量
        u1 = to_vec(A, B)
        L1 = norm(u1)
        u1 = (u1[0] / L1, u1[1] / L1)

        u2 = to_vec(B, C)
        L2 = norm(u2)
        u2 = (u2[0] / L2, u2[1] / L2)

        # 2) 计算指向内侧的法向量 n1, n2
        #    初始法向量为 90° 旋转
        n1 = (-u1[1], u1[0])
        # 如果这个方向朝外，则翻转
        if dot(n1, to_vec(B, C)) < 0:
            n1 = (-n1[0], -n1[1])

        n2 = (-u2[1], u2[0])
        if dot(n2, to_vec(B, A)) < 0:
            n2 = (-n2[0], -n2[1])

        # 3) 构造两条平移后的直线：
        #    L1_offset: 过 A + R*n1，方向 u1
        #    L2_offset: 过 B + R*n2，方向 u2
        P1 = add((A.x, A.y), scale(n1, R))
        P2 = add((B.x, B.y), scale(n2, R))

        # 求两直线交点 O_blend = P1 + t1 * u1
        denom = cross(u1, u2)
        if abs(denom) < 1e-9:
            raise ValueError("AB 与 BC 平移线近似平行，无法确定圆心")
        t1 = cross((P2[0] - P1[0], P2[1] - P1[1]), u2) / denom
        O_blend = (P1[0] + t1 * u1[0], P1[1] + t1 * u1[1])

        # 4) 在原始直线 AB 上，求与 O_blend 距离 = R 的点 N
        #    参数化： N(s) = A + s * u1
        #    要求 || N(s) - O_blend ||^2 = R^2
        #    ⇔ ||A + s u1 - O||^2 = R^2  → 二次方程 as^2 + bs + c = 0
        w = (A.x - O_blend[0], A.y - O_blend[1])
        a = dot(u1, u1)
        b = 2 * dot(u1, w)
        c = dot(w, w) - R * R

        disc = b * b - 4 * a * c
        if disc < 0:
            raise ValueError("理论上不应发生：AB 与圆无交点")

        sqrt_d = math.sqrt(disc)
        s1 = (-b + sqrt_d) / (2 * a)
        s2 = (-b - sqrt_d) / (2 * a)

        # 5) 从两个解中选一个：应当是 “离 A 更近、且位于 AB 段上” 的那个
        candidates = []
        for s in (s1, s2):
            if 0 <= s <= L1:
                nx, ny = A.x + s * u1[0], A.y + s * u1[1]
                candidates.append((nx, ny))
        if not candidates:
            raise ValueError("在 AB 段上未找到合适的 N 点")

        # 若两个都符合，取距离 B 更远（提前转弯）的那个
        def dist2(p, Q):
            return (p[0] - Q.x) ** 2 + (p[1] - Q.y) ** 2

        N = max(candidates, key=lambda p: dist2(p, B))
        return Point(*N)


    def _generate_arc_path(self, start_point, end_point, center):
        """生成圆弧路径点"""
        theta_start = math.atan2(start_point.y - center.y, start_point.x - center.x)
        theta_end = math.atan2(end_point.y - center.y, end_point.x - center.x)

        # 确保角度递增方向正确
        if theta_end < theta_start:
            theta_end += 2 * math.pi

        theta_values = np.linspace(theta_start, theta_end, num=8)
        return [
            Point(
                center.x + 2 * self.turn_radius * math.cos(theta),
                center.y + 2 * self.turn_radius * math.sin(theta)
            ) for theta in theta_values
        ]

    def compute_blend_center(self, A: Point, B: Point, C: Point, R: float) -> Point:
        """
        计算平滑转弯圆心 O4
        """
        # 复用 compute_blend_start 求 O_blend
        # 直接 compute offset lines intersection
        # 确保所有计算使用 numpy.float64
        def to_vec(P, Q):
            return np.float64(Q.x - P.x), np.float64(Q.y - P.y)
        def norm(v):
            return np.hypot(np.float64(v[0]), np.float64(v[1]))
        def scale(v, s):
            return np.float64(v[0] * s), np.float64(v[1] * s)
        def add(u, v):
            return np.float64(u[0] + v[0]), np.float64(u[1] + v[1])
        def dot(u, v):
            return np.float64(u[0] * v[0] + u[1] * v[1])
        def cross(u, v):
            return np.float64(u[0] * v[1] - u[1] * v[0])

        u1=to_vec(A,B);L1=norm(u1);u1=(u1[0]/L1,u1[1]/L1)
        u2=to_vec(B,C);L2=norm(u2);u2=(u2[0]/L2,u2[1]/L2)
        n1=(-u1[1],u1[0]);
        if dot(n1,to_vec(B,C))<0:n1=(-n1[0],-n1[1])
        n2=(-u2[1],u2[0]);
        if dot(n2,to_vec(B,A))<0:n2=(-n2[0],-n2[1])
        P1=add((A.x,A.y),scale(n1,R))
        P2=add((B.x,B.y),scale(n2,R))
        denom=cross(u1,u2)
        if abs(denom)<1e-9:raise ValueError("Parallel")
        t1=cross((P2[0]-P1[0],P2[1]-P1[1]),u2)/denom
        O_blend=(P1[0]+t1*u1[0],P1[1]+t1*u1[1])
        return Point(*O_blend)
